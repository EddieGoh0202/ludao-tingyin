import { XIAMEN_TONE_CONTOUR, tonesOf } from '../learning/tailo';

export type Sfx = 'click' | 'lock' | 'ok' | 'bad' | 'collect' | 'dice' | 'unlock';

/**
 * 音频管理：收音机杂音 + 人声（占位合成）+ 低通滤波。
 * 正式版本中 playPhrase 应改为播放母语者录音（SoundEntry.audio）。
 */
class AudioManagerImpl {
  private ctx: AudioContext | null = null;
  private master!: GainNode;
  private noiseGain!: GainNode;
  private voiceGain!: GainNode;
  private voiceFilter!: BiquadFilterNode;
  private cleanBus!: GainNode;
  private noiseSrc: AudioBufferSourceNode | null = null;
  private loopKey: string | null = null;
  private loopTimer: number | null = null;
  private volume = 0.8;

  get ready(): boolean {
    return this.ctx !== null;
  }

  /** 必须在用户手势中调用 */
  init(): void {
    if (this.ctx) {
      void this.ctx.resume();
      return;
    }
    const Ctx = window.AudioContext ?? (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (!Ctx) return;
    const ctx = new Ctx();
    this.ctx = ctx;
    this.master = ctx.createGain();
    this.master.gain.value = this.volume;
    this.master.connect(ctx.destination);

    this.noiseGain = ctx.createGain();
    this.noiseGain.gain.value = 0;
    const noiseBand = ctx.createBiquadFilter();
    noiseBand.type = 'bandpass';
    noiseBand.frequency.value = 2200;
    noiseBand.Q.value = 0.4;
    this.noiseGain.connect(noiseBand).connect(this.master);

    this.voiceFilter = ctx.createBiquadFilter();
    this.voiceFilter.type = 'lowpass';
    this.voiceFilter.frequency.value = 400;
    this.voiceGain = ctx.createGain();
    this.voiceGain.gain.value = 0;
    this.voiceFilter.connect(this.voiceGain).connect(this.master);

    this.cleanBus = ctx.createGain();
    this.cleanBus.gain.value = 1;
    this.cleanBus.connect(this.master);
  }

  setVolume(v: number): void {
    this.volume = v;
    if (this.ctx) this.master.gain.setTargetAtTime(v, this.ctx.currentTime, 0.05);
  }

  radioOn(on: boolean): void {
    const ctx = this.ctx;
    if (!ctx) return;
    if (on && !this.noiseSrc) {
      const len = ctx.sampleRate * 2;
      const buf = ctx.createBuffer(1, len, ctx.sampleRate);
      const data = buf.getChannelData(0);
      for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
      const src = ctx.createBufferSource();
      src.buffer = buf;
      src.loop = true;
      src.connect(this.noiseGain);
      src.start();
      this.noiseSrc = src;
    } else if (!on) {
      this.noiseSrc?.stop();
      this.noiseSrc = null;
      this.setVoiceLoop(null);
      this.setMix(0, 0, 400);
    }
  }

  /** 杂音音量、人声音量、低通截止频率 */
  setMix(noise: number, voice: number, cutoff: number): void {
    const ctx = this.ctx;
    if (!ctx) return;
    const t = ctx.currentTime;
    this.noiseGain.gain.setTargetAtTime(noise * 0.35, t, 0.05);
    this.voiceGain.gain.setTargetAtTime(voice, t, 0.05);
    this.voiceFilter.frequency.setTargetAtTime(cutoff, t, 0.05);
  }

  /** 在收音机通道中循环播放某句话（经过滤波） */
  setVoiceLoop(tailo: string | null): void {
    if (tailo === this.loopKey) return;
    if (this.loopTimer !== null) window.clearTimeout(this.loopTimer);
    this.loopTimer = null;
    this.loopKey = tailo;
    if (!tailo || !this.ctx) return;
    const loop = () => {
      if (this.loopKey !== tailo) return;
      const dur = this.synthPhrase(tailo, this.voiceFilter);
      this.loopTimer = window.setTimeout(loop, (dur + 0.8) * 1000);
    };
    loop();
  }

  /** 清晰播放一句（挑战、收集簿用），返回时长（秒） */
  playPhrase(tailo: string, rate = 1): number {
    if (!this.ctx) return 0;
    void this.ctx.resume();
    return this.synthPhrase(tailo, this.cleanBus, rate);
  }

  /** 播放单个调类的示意音 */
  playTone(tone: number): number {
    if (!this.ctx) return 0;
    return this.synthSyllables([tone], this.cleanBus, 1.4);
  }

  private synthPhrase(tailo: string, dest: AudioNode, rate = 1): number {
    const tones = tonesOf(tailo);
    return this.synthSyllables(tones.length ? tones : [1], dest, rate);
  }

  /** 占位合成：按厦门腔调值生成音高轮廓（不是真人发音） */
  private synthSyllables(tones: number[], dest: AudioNode, rate: number): number {
    const ctx = this.ctx!;
    let t = ctx.currentTime + 0.05;
    const pitch = (level: number) => 130 * Math.pow(2, (level - 1) / 4);
    for (const tone of tones) {
      const c = XIAMEN_TONE_CONTOUR[tone] ?? XIAMEN_TONE_CONTOUR[1];
      const dur = (c.short ? 0.16 : 0.32) * rate;
      const osc = ctx.createOscillator();
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(pitch(c.from), t);
      osc.frequency.linearRampToValueAtTime(pitch(c.to), t + dur);
      const formant = ctx.createBiquadFilter();
      formant.type = 'bandpass';
      formant.frequency.value = 900;
      formant.Q.value = 1.2;
      const env = ctx.createGain();
      env.gain.setValueAtTime(0, t);
      env.gain.linearRampToValueAtTime(0.9, t + 0.03);
      env.gain.setValueAtTime(0.9, t + dur - (c.short ? 0.02 : 0.08));
      env.gain.linearRampToValueAtTime(0, t + dur);
      osc.connect(formant).connect(env).connect(dest);
      osc.start(t);
      osc.stop(t + dur + 0.02);
      t += dur + 0.07 * rate;
    }
    return t - ctx.currentTime;
  }

  sfx(kind: Sfx): void {
    const ctx = this.ctx;
    if (!ctx) return;
    const seq: Record<Sfx, Array<[number, number]>> = {
      click: [[880, 0.04]],
      lock: [[660, 0.08], [990, 0.12]],
      ok: [[784, 0.08], [1046, 0.14]],
      bad: [[220, 0.18]],
      collect: [[523, 0.08], [659, 0.08], [784, 0.08], [1046, 0.2]],
      dice: [[300, 0.03], [420, 0.03], [360, 0.03]],
      unlock: [[392, 0.1], [523, 0.1], [784, 0.25]],
    };
    let t = ctx.currentTime;
    for (const [f, d] of seq[kind]) {
      const osc = ctx.createOscillator();
      osc.type = 'square';
      osc.frequency.value = f;
      const g = ctx.createGain();
      g.gain.setValueAtTime(0.12, t);
      g.gain.exponentialRampToValueAtTime(0.001, t + d);
      osc.connect(g).connect(this.cleanBus);
      osc.start(t);
      osc.stop(t + d);
      t += d;
    }
  }
}

export const AudioManager = new AudioManagerImpl();

    return UIState.modalOpen || this.challengeActive || this.time.now - this.lastModalAt < 200;
  }

  private walkTo(x: number, y: number): void {
    this.target = {
      x: Phaser.Math.Clamp(x, MIN_X, MAX_X),
      y: Phaser.Math.Clamp(y, MIN_Y, MAX_Y),
    };
  }

  private distanceTo(h: HotspotView): number {
    return Math.hypot(h.x - this.player.x, (h.y - this.player.y) * 0.6);
  }

  private interact(h: HotspotView): void {
    this.target = null;
    this.pendingTalk = null;
    AudioManager.init();
    AudioManager.sfx('click');
    this.player.setFlipX(h.x < this.player.x);
    if (h.kind === 'knowledge' && h.know) {
      KnowledgeSystem.open(h.know, () => {
        this.lastModalAt = this.time.now;
        this.refreshMark(h);
      });
      return;
    }
    const def = h.def!;
    switch (def.action) {
      case 'npc': {
        const name = def.label.replace(/[（(].*$/, '');
        const lines: DialogueLine[] = def.dialogue?.length
          ? def.dialogue.map((d) => ({ speaker: d.speaker ?? name, text: d.text, tailo: d.tailo }))
          : [{ speaker: name, text: def.npcLine ?? '……' }];
        DialogueSystem.show(lines, undefined, {
          bg: sceneBackdrop(this.sc.id),
          portrait: def.sprite ? portrait(def.sprite) : null,
          playerPortrait: portrait('player'),
        });
        break;
      }
      case 'minigame:bargain':
        openBargain();
        break;
      case 'minigame:bobing':
        openBobing();
        break;
      default:
        break;
    }
  }

  private onLocked(soundId: string): void {
    if (this.challengeActive) return;
    this.challengeActive = true;
    this.target = null;
    this.radio.setPaused(true);
    runChallenge(
      soundId,
      (r) => {
        CollectionSystem.collect(soundId, r.stars);
        this.challengeActive = false;
        this.lastModalAt = this.time.now;
        this.radio.setPaused(false);
        QuestSystem.evaluate();
        this.refreshWaves();
        this.hud.refresh();
      },
      () => {
        // 放弃挑战：恢复收音，5 秒内不再锁定同一段声音
        this.challengeActive = false;
        this.lastModalAt = this.time.now;
        this.radio.snooze(soundId, 5);
        this.radio.setPaused(false);
      },
    );
  }

  private applyTint(): void {
    const t = TIME_TINT[TimeSystem.current];
    this.tint.setFillStyle(t.color, t.alpha);
  }

  /** 当前时段可听到、尚未收集的声源显示「声波」提示 */
  private refreshWaves(): void {
    const audible = Content.audibleIn(this.sc.id, TimeSystem.current).filter((s) => !StateStore.isCollected(s.id));
    const keep = new Set(audible.map((s) => s.id));
    for (const [id, img] of this.waves) {
      if (!keep.has(id)) {
        this.tweens.killTweensOf(img);
        img.destroy();
        this.waves.delete(id);
      }
    }
    for (const s of audible) {
      if (this.waves.has(s.id)) continue;
      const img = this.add.image(s.pos.x, s.pos.y - 14, 'wave').setDepth(15).setAlpha(0.35);
      this.tweens.add({
        targets: img,
        scale: { from: 0.7, to: 1.15 },
        alpha: { from: 0.45, to: 0.1 },
        duration: 1100 + (s.frequency % 1) * 600,
        repeat: -1,
        ease: 'Sine.easeOut',
      });
      this.waves.set(s.id, img);
    }
  }

  update(time: number, delta: number): void {
    const dt = Math.min(delta, 50) / 1000;
    const modal = UIState.modalOpen;
    if (modal) this.lastModalAt = time;

    // 弹窗 / 对话期间暂停收音（挑战由 onLocked 单独暂停）
    const shouldPause = modal && !this.challengeActive;
    if (shouldPause !== this.uiPaused) {
      this.uiPaused = shouldPause;
      if (!this.challengeActive) this.radio.setPaused(shouldPause);
    }

    const blocked = this.inputBlocked();
    if (!blocked) {
      this.handleTuning(dt);
      this.handleMovement(dt);
      if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
        this.scene.start('WorldMap');
        return;
      }
    } else {
      this.setWalking(false);
      // 消耗掉弹窗期间的按键，避免关闭后误触发
      Phaser.Input.Keyboard.JustDown(this.keys.space);
      Phaser.Input.Keyboard.JustDown(this.keys.enter);
      Phaser.Input.Keyboard.JustDown(this.keys.q);
      Phaser.Input.Keyboard.JustDown(this.keys.e);
      Phaser.Input.Keyboard.JustDown(this.keys.m);
    }

    // 最近的可交互对象
    let best: HotspotView | null = null;
    let bestD = Infinity;
    for (const h of this.hotspots) {
      const d = this.distanceTo(h) / this.rangeOf(h);
      if (d < bestD) {
        bestD = d;
        best = h;
      }
    }
    this.nearest = best && bestD <= 1 ? best : null;

    if (this.pendingTalk && this.distanceTo(this.pendingTalk) <= this.rangeOf(this.pendingTalk) && !blocked) {
      this.interact(this.pendingTalk);
    } else if (!blocked && this.nearest) {
      const sp = Phaser.Input.Keyboard.JustDown(this.keys.space);
      const en = Phaser.Input.Keyboard.JustDown(this.keys.enter);
      if (sp || en) this.interact(this.nearest);
    } else if (!blocked) {
      Phaser.Input.Keyboard.JustDown(this.keys.space);
      Phaser.Input.Keyboard.JustDown(this.keys.enter);
    }

    // 收音机
    const sig = this.radio.update(
      dt,
      this.player.x,
      this.player.y,
      Content.audibleIn(this.sc.id, TimeSystem.current),
      (id) => StateStore.isCollected(id),
    );
    this.panel.render(sig);
    for (const [id, img] of this.waves) img.setTint(id === sig.nearestId ? 0xffffff : 0xd9a441);

    this.updateLabels();
  }

  private handleTuning(dt: number): void {
    const k = this.keys;
    const step = k.shift.isDown ? 1 : 0.1;
    if (Phaser.Input.Keyboard.JustDown(k.q)) {
      this.radio.tune(-step);
      this.tuneHold = 0;
    }
    if (Phaser.Input.Keyboard.JustDown(k.e)) {
      this.radio.tune(step);
      this.tuneHold = 0;
    }
    const dir = (k.e.isDown ? 1 : 0) - (k.q.isDown ? 1 : 0);
    if (dir === 0) {
      this.tuneHold = 0;
      return;
    }
    // 按住 0.35 秒后连续调频
    const prev = this.tuneHold;
    this.tuneHold += dt;
    if (this.tuneHold > 0.35) {
      const ticks = Math.floor((this.tuneHold - 0.35) / 0.06) - Math.floor(Math.max(0, prev - 0.35) / 0.06);
      if (ticks > 0) this.radio.tune(dir * step * ticks);
    }
  }

  private handleMovement(dt: number): void {
    const k = this.keys;
    let vx = (k.right.isDown || k.d.isDown ? 1 : 0) - (k.left.isDown || k.a.isDown ? 1 : 0);
    let vy = (k.down.isDown || k.s.isDown ? 1 : 0) - (k.up.isDown || k.w.isDown ? 1 : 0);

    if (vx !== 0 || vy !== 0) {
      this.target = null;
      this.pendingTalk = null;
    } else if (this.target) {
      const dx = this.target.x - this.player.x;
      const dy = this.target.y - this.player.y;
      const d = Math.hypot(dx, dy);
      if (d < 1.5) {
        this.target = null;
      } else {
        vx = dx / d;
        vy = dy / d;
      }
    }

    if (vx === 0 && vy === 0) {
      this.setWalking(false);
      return;
    }
    const len = Math.hypot(vx, vy);
    const step = SPEED * dt;
    let nx = this.player.x + (vx / len) * step;
    let ny = this.player.y + (vy / len) * step * 0.75;
    if (this.target) {
      // 不越过目标点
      if (Math.abs(this.target.x - this.player.x) < Math.abs(nx - this.player.x)) nx = this.target.x;
      if (Math.abs(this.target.y - this.player.y) < Math.abs(ny - this.player.y)) ny = this.target.y;
    }
    this.player.x = Phaser.Math.Clamp(nx, MIN_X, MAX_X);
    this.player.y = Phaser.Math.Clamp(ny, MIN_Y, MAX_Y);
    this.player.setDepth(depthOf(this.player.y));
    if (vx !== 0) this.player.setFlipX(vx < 0);
    this.setWalking(true, dt);
  }

  private setWalking(walking: boolean, dt = 0): void {
    if (!walking) {
      if (this.player.texture.key !== 'player') this.player.setTexture('player');
      this.walkClock = 0;
      return;
    }
    this.walkClock += dt;
    const seq = this.textures.exists('player-walk2') ? ['player-walk', 'player', 'player-walk2', 'player'] : ['player-walk', 'player'];
    const frame = seq[Math.floor(this.walkClock / 0.14) % seq.length];
    if (this.player.texture.key !== frame) this.player.setTexture(frame);
  }

  private updateLabels(): void {
    const cam = this.cameras.main;
    const showHint = !!this.nearest && !this.inputBlocked();
    for (const h of this.hotspots) {
      const sx = h.x - cam.scrollX;
      const sy = h.y - h.sprite.displayHeight - 3 - cam.scrollY;
      // 科普点平时只显示头顶标记，靠近时才出现名字
      const visible = sx > -40 && sx < 520 && h.kind === 'hotspot';
      h.tag.classList.toggle('hidden', !visible || (showHint && h === this.nearest));
      if (visible) Stage.place(h.tag, sx, sy);
    }
    this.hintTag.classList.toggle('hidden', !showHint);
    const n = showHint ? this.nearest : null;
    if (n && n.kind === 'knowledge' && n.know) {
      const k = n.know;
      const html = `<b>${k.title}</b>${KnowledgeSystem.isSeen(k.id) ? ' <em>已读</em>' : ''}<br>${k.summary}`;
      if (this.bubble.dataset.id !== k.id) {
        this.bubble.dataset.id = k.id;
        this.bubble.innerHTML = html;
      }
      this.bubble.classList.remove('hidden');
      Stage.place(this.bubble, Phaser.Math.Clamp(n.x - cam.scrollX, 70, 410), n.y - n.sprite.displayHeight - 24 - cam.scrollY);
    } else {
      this.bubble.classList.add('hidden');
      this.bubble.dataset.id = '';
    }
    if (n) {
      const verb = n.kind === 'knowledge' ? '了解' : n.def?.action === 'npc' ? '交谈' : '开始';
      const text = `空格 ${verb}：${n.label}`;
      if (this.hintTag.textContent !== text) this.hintTag.textContent = text;
      this.hintTag.classList.toggle('know', n.kind === 'knowledge');
      Stage.place(this.hintTag, n.x - cam.scrollX, n.y - n.sprite.displayHeight - (n.kind === 'knowledge' ? 14 : 3) - cam.scrollY);
    }
  }

  private cleanup(): void {
    DialogueSystem.dismiss();
    this.offs.forEach((f) => f());
    this.offs = [];
    this.radio?.setOn(false);
    AudioManager.radioOn(false);
    this.hud?.destroy();
    this.panel?.destroy();
    this.labelsRoot?.remove();
    Stage.clear('labels');
    this.input.removeAllListeners();
    this.input.keyboard?.removeAllKeys(false);
  }
}

import { AudioManager } from '../../core/AudioManager';
import { Stage, UIState, el } from '../../ui/dom';

export interface DialogueLine {
  speaker?: string;
  text: string;
  /** 有台罗时显示「▶」按钮，可播放占位合成音 */
  tailo?: string;
  /** 该句的立绘（dataURL）；未提供则沿用对话选项里的立绘 */
  portrait?: string | null;
}

export interface DialogueOptions {
  /** 全屏背景插画 dataURL：提供时进入「视觉小说」式对话 */
  bg?: string;
  /** 对方立绘 dataURL */
  portrait?: string | null;
  /** 玩家立绘 dataURL（speaker 为「我」时显示在左侧） */
  playerPortrait?: string | null;
}

/**
 * 底部对话框：逐字显示，点击 / 空格 / 回车 → 若未打完则补全，否则下一句。
 * 对话期间计入 UIState.modalDepth，场景不响应移动。
 */
class DialogueSystemImpl {
  private queue: DialogueLine[] = [];
  private box: HTMLDivElement | null = null;
  private textEl!: HTMLDivElement;
  private speakerEl!: HTMLDivElement;
  private playBtn!: HTMLButtonElement;
  private scrim: HTMLDivElement | null = null;
  private portraitEl: HTMLImageElement | null = null;
  private opts: DialogueOptions = {};
  private full = '';
  private shown = 0;
  private timer: number | null = null;
  private onDone: (() => void) | null = null;
  private keyHandler = (e: KeyboardEvent) => {
    if (e.key === ' ' || e.key === 'Enter' || e.key === 'e' || e.key === 'E') {
      e.preventDefault();
      e.stopPropagation();
      this.advance();
    }
  };

  get active(): boolean {
    return this.box !== null;
  }

  /** 切换场景时强制关闭：清空队列、释放 modalDepth，不触发 onDone */
  dismiss(): void {
    if (!this.box) return;
    this.queue = [];
    this.onDone = null;
    this.close();
  }

  show(lines: DialogueLine[] | DialogueLine, onDone?: () => void, opts: DialogueOptions = {}): void {
    const arr = Array.isArray(lines) ? lines : [lines];
    if (!arr.length) {
      onDone?.();
      return;
    }
    if (this.box) {
      // 已在对话中：追加到队列，回调串联
      this.queue.push(...arr);
      const prev = this.onDone;
      this.onDone = () => {
        prev?.();
        onDone?.();
      };
      return;
    }
    this.queue = [...arr];
    this.onDone = onDone ?? null;
    this.opts = opts;
    this.open();
    this.next();
  }

  private open(): void {
    const layer = Stage.layer('dialogue');
    if (this.opts.bg) {
      const bg = el('div', { class: 'dlg-bg' });
      bg.style.backgroundImage = `url(${this.opts.bg})`;
      this.portraitEl = el('img', { class: 'dlg-portrait hidden' }) as HTMLImageElement;
      this.portraitEl.alt = '';
      this.scrim = el('div', { class: 'dlg-scrim' }, [bg, el('div', { class: 'dlg-shade' }), this.portraitEl]);
      layer.append(this.scrim);
    }
    this.speakerEl = el('div', { class: 'dlg-speaker' });
    this.textEl = el('div', { class: 'dlg-text' });
    this.playBtn = el('button', { class: 'dlg-play hidden', text: '▶', title: '播放' });
    this.playBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const t = this.playBtn.dataset.tailo;
      if (t) AudioManager.playPhrase(t);
    });
    const more = el('div', { class: 'dlg-more', text: '▼' });
    this.box = el('div', { class: `dlg-box ${this.opts.bg ? 'vn' : ''}` }, [this.speakerEl, this.textEl, this.playBtn, more]);
    this.box.addEventListener('click', () => this.advance());
    this.scrim?.addEventListener('click', () => this.advance());
    layer.append(this.box);
    UIState.modalDepth += 1;
    window.addEventListener('keydown', this.keyHandler, true);
  }

  private close(): void {
    if (this.timer !== null) window.clearInterval(this.timer);
    this.timer = null;
    this.box?.remove();
    this.box = null;
    this.scrim?.remove();
    this.scrim = null;
    this.portraitEl = null;
    this.opts = {};
    UIState.modalDepth = Math.max(0, UIState.modalDepth - 1);
    window.removeEventListener('keydown', this.keyHandler, true);
    const cb = this.onDone;
    this.onDone = null;
    cb?.();
  }

  private next(): void {
    const line = this.queue.shift();
    if (!line) {
      this.close();
      return;
    }
    this.speakerEl.textContent = line.speaker ?? '';
    this.speakerEl.classList.toggle('hidden', !line.speaker);
    if (this.portraitEl) {
      const isPlayer = line.speaker === '我';
      const src = line.portrait !== undefined ? line.portrait : isPlayer ? this.opts.playerPortrait : this.opts.portrait;
      this.portraitEl.classList.toggle('hidden', !src);
      this.portraitEl.classList.toggle('left', isPlayer);
      if (src && this.portraitEl.src !== src) {
        this.portraitEl.src = src;
        this.portraitEl.classList.remove('enter');
        void this.portraitEl.offsetWidth;
        this.portraitEl.classList.add('enter');
      }
    }
    this.playBtn.classList.toggle('hidden', !line.tailo);
    this.playBtn.dataset.tailo = line.tailo ?? '';
    this.full = line.text;
    this.shown = 0;
    this.textEl.textContent = '';
    if (this.timer !== null) window.clearInterval(this.timer);
    this.timer = window.setInterval(() => {
      this.shown += 1;
      this.textEl.textContent = this.full.slice(0, this.shown);
      if (this.shown >= this.full.length && this.timer !== null) {
        window.clearInterval(this.timer);
        this.timer = null;
      }
    }, 28);
  }

  private advance(): void {
    if (!this.box) return;
    AudioManager.sfx('click');
    if (this.timer !== null) {
      window.clearInterval(this.timer);
      this.timer = null;
      this.textEl.textContent = this.full;
      return;
    }
    this.next();
  }
}

export const DialogueSystem = new DialogueSystemImpl();
